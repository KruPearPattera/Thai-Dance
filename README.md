  function fieldHTML([key,label,max]){return `<div class="field"><label for="${key}">${label}</label><input id="${key}" class="canva-input" type="number" min="0" max="${max}" step="1"></div>`}
  $('term-one-fields').innerHTML=fields.slice(0,6).map(fieldHTML).join('');
  $('term-two-fields').innerHTML=fields.slice(6).map(fieldHTML).join('');
  function openForm(s=null){
    if(!ready){message('ยังไม่พร้อมบันทึกข้อมูล กรุณาลองอีกครั้ง');return}
    editingNumber=s?.student_number??null;$('form-title').textContent=s?'แก้ไขข้อมูลนักเรียน':'เพิ่มนักเรียน';
    $('student-number').value=s?.student_number??'';$('student-name').value=s?.student_name??'';
    for(const [key] of fields)$(key).value=s?.[key]??'';
    for(const key of ['overall_total','grade_level','student_note','source_level'])$(key.replaceAll('_','-')).value=s?.[key]??'';
    $('attendance').value=s?.attendance??'';
    $('form-message').textContent='';$('form-overlay').classList.add('open');$('student-number').focus();
  }
  function closeForm(){$('form-overlay').classList.remove('open');editingNumber=null}
  $('add-button').onclick=()=>openForm();$('close-form').onclick=closeForm;$('cancel-form').onclick=closeForm;
  $('form-overlay').addEventListener('click',e=>{if(e.target===$('form-overlay'))closeForm()});
  document.addEventListener('keydown',e=>{if(e.key==='Escape'&&$('form-overlay').classList.contains('open'))closeForm()});
  $('search-input').addEventListener('input',render);
  $('student-table').addEventListener('click',e=>{const row=e.target.closest('tr[data-number]');if(row){selectedNumber=Number(row.dataset.number);render()}});
  $('student-table').addEventListener('keydown',e=>{if((e.key==='Enter'||e.key===' ')&&e.target.matches('tr[data-number]')){e.preventDefault();selectedNumber=Number(e.target.dataset.number);render()}});
  $('student-form').addEventListener('submit',async e=>{
    e.preventDefault();
    const number=Number($('student-number').value),name=$('student-name').value.trim();
    if(!Number.isInteger(number)||number<1||number>999){$('form-message').textContent='กรุณากรอกเลขที่ระหว่าง 1–999';return}
    if(rows().some(s=>s.student_number===number&&s.student_number!==editingNumber)){$('form-message').textContent='เลขที่นี้มีนักเรียนอยู่แล้ว';return}
    if(!saved.some(s=>Number(s.student_number)===editingNumber)&&saved.length>=999){$('form-message').textContent='มีข้อมูลครบ 999 รายการแล้ว';return}
    const detail={};
    for(const [key,,max] of fields){
      const raw=$(key).value;
      if(raw!==''&&(!Number.isInteger(Number(raw))||Number(raw)<0||Number(raw)>max)){$('form-message').textContent='กรุณาตรวจสอบคะแนนรายด้านให้เป็นจำนวนเต็มตามคะแนนเต็ม';return}
      detail[key]=raw===''?'':Number(raw);
    }
    const rawTotal=$('overall-total').value;
    if(rawTotal!==''&&(!Number.isInteger(Number(rawTotal))||Number(rawTotal)<0||Number(rawTotal)>100)){$('form-message').textContent='คะแนนรวมต้องอยู่ระหว่าง 0–100';return}
    detail.overall_total=rawTotal===''?'':Number(rawTotal);
    detail.grade_level=$('grade-level').value.trim();
    detail.student_note=$('student-note').value.trim();
    detail.source_level=$('source-level').value.trim();
    const attendanceRaw=$('attendance').value;
    if(attendanceRaw!==''&&(!Number.isInteger(Number(attendanceRaw))||Number(attendanceRaw)<0||Number(attendanceRaw)>40)){$('form-message').textContent='เวลาเรียนต้องอยู่ระหว่าง 0–40';return}
    detail.attendance=attendanceRaw===''?'':Number(attendanceRaw);
    detail.review_status=(!name||Object.values(detail).some(v=>v===''))?'รอตรวจสอบ':'';
    const old=saved.find(s=>Number(s.student_number)===editingNumber);
    const record={student_name:name,student_number:number,performance_score:0,rhythm_score:0,participation_score:0,created_at:old?.created_at||new Date().toISOString(),record_json:JSON.stringify(detail),review_status:detail.review_status};
    const btn=$('save-button');btn.disabled=true;btn.textContent='กำลังบันทึก...';$('form-message').textContent='';
    try{
      const result=old?await window.dataSdk.update({...old,...record}):await window.dataSdk.create(record);
      if(result.isOk){selectedNumber=number;closeForm();message('บันทึกข้อมูลเรียบร้อยแล้ว',true)}
      else $('form-message').textContent='บันทึกไม่สำเร็จ กรุณาลองอีกครั้ง';
    }catch(err){$('form-message').textContent='บันทึกไม่สำเร็จ กรุณาลองอีกครั้ง'}
    finally{btn.disabled=false;btn.textContent='บันทึกคะแนน'}
  });
  async function start(){
    render();lucide.createIcons();
    try{
      const result=await window.dataSdk.init({onDataChanged(data){saved=data;render()}});
      if(result.isOk)ready=true;else message('ไม่สามารถโหลดข้อมูลที่บันทึกไว้ได้ กรุณาลองเปิดหน้าอีกครั้ง');
    }catch(err){message('ไม่สามารถโหลดข้อมูลที่บันทึกไว้ได้ กรุณาลองเปิดหน้าอีกครั้ง')}
  }
  start();
